🔐 Smart Contracts - Cross-Border Escrow

Solidity contracts for secure cross-border transactions

CrossBorderEscrow.sol - Main Contract 
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.20;

import "@openzeppelin/contracts/token/ERC20/IERC20.sol";
import "@openzeppelin/contracts/utils/ReentrancyGuard.sol";

contract CrossBorderEscrow is ReentrancyGuard {
    enum State { Created, Funded, Disbursed, Disputed, Refunded }

    struct EscrowTransaction {
        address buyer;
        address seller;
        IERC20 token;
        uint256 amount;
        State state;
    }

    mapping(bytes32 => EscrowTransaction) public escrows;

    event EscrowCreated(bytes32 indexed escrowId, address buyer, address seller, uint256 amount);
    event EscrowFunded(bytes32 indexed escrowId);
    event EscrowReleased(bytes32 indexed escrowId);

    function createAndFundEscrow(
        bytes32 escrowId,
        address seller,
        IERC20 token,
        uint256 amount
    ) external nonReentrant {
        require(escrows[escrowId].buyer == address(0), "Escrow exists");
        require(amount > 0, "Amount must be > 0");

        escrows[escrowId] = EscrowTransaction({
            buyer: msg.sender,
            seller: seller,
            token: token,
            amount: amount,
            state: State.Funded
        });

        require(token.transferFrom(msg.sender, address(this), amount), "Transfer failed");

        emit EscrowCreated(escrowId, msg.sender, seller, amount);
        emit EscrowFunded(escrowId);
    }

    function releaseFunds(bytes32 escrowId) external nonReentrant {
        EscrowTransaction storage escrow = escrows[escrowId];
        require(msg.sender == escrow.buyer, "Only buyer can release funds");
        require(escrow.state == State.Funded, "Invalid state");

        escrow.state = State.Disbursed;
        require(escrow.token.transfer(escrow.seller, escrow.amount), "Payout failed");

        emit EscrowReleased(escrowId);
    }
}

🔔 Alert Response Procedures

🚨 Critical Alerts: Require response within 5 minutes

🔴 CRITICAL: Health Check Failing

# Test directly
curl -v https://api.eascrow.io/api/health

# Check app status
pm2 status

# If not online, restart
pm2 restart escrow-api

🔴 CRITICAL: Database Down

# Check RDS status
aws rds describe-db-instances \
  --db-instance-identifier escrow-prod-db

# Trigger reboot
aws rds reboot-db-instance \
  --db-instance-identifier escrow-prod-db

🚀 Complete Deployment Guide

✅ Multiple Deployment Options: Choose the one that works best for you

🔵 AWS S3 Deployment (Recommended)

Step 1: Create S3 Bucket 
aws s3 mb s3://escrow-docs --region us-east-1

# Enable static website hosting
aws s3 website s3://escrow-docs \
  --index-document index.html \
  --error-document index.html

# Make bucket public
aws s3api put-bucket-policy --bucket escrow-docs \
  --policy '{
    "Version": "2012-10-17",
    "Statement": [{
      "Effect": "Allow",
      "Principal": "*",
      "Action": "s3:GetObject",
      "Resource": "arn:aws:s3:::escrow-docs/*"
    }]
  }'
Step 2: Upload Files 
aws s3 cp index.html s3://escrow-docs/index.html \
  --content-type text/html
